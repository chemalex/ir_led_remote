cd /home/alex/ir_send_rcv/IRMP

sudo apt update
sudo apt install gcc-avr avr-libc binutils-avr avrdude

avr-g++ -std=gnu++11 \
  -mmcu=atmega8 \
  -DF_CPU=12000000UL \
  -Os -ffunction-sections -fdata-sections \
  -fno-exceptions -fno-rtti \
  -Isrc \
  examples/ATmega8UARTReceiver/ATmega8UARTReceiver.cpp \
  -Wl,--gc-sections \
  -o examples/ATmega8UARTReceiver/receiver.elf


  avr-size -C --mcu=atmega8 examples/ATmega8UARTReceiver/receiver.elf

  avr-objcopy -O ihex -R .eeprom \
  examples/ATmega8UARTReceiver/receiver.elf \
  examples/ATmega8UARTReceiver/receiver.hex

  avrdude -c usbasp -p m8 \
  -U flash:w:examples/ATmega8UARTReceiver/receiver.hex:i
